//==================================================
// FUNCTIONS
//==================================================
f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(x, hi))

f_safe_div(float num, float den, float fallback) =>
    den == 0.0 or na(den) ? fallback : num / den

f_fmt(float x) =>
    str.tostring(x, format.mintick)

f_pos(string p) =>
    p == "TOP_LEFT" ? position.top_left : p == "BOTTOM_RIGHT" ? position.bottom_right : p == "BOTTOM_LEFT" ? position.bottom_left : position.top_right

f_size(string s) =>
    s == "tiny" ? size.tiny : s == "normal" ? size.normal : size.small

f_bool_txt(bool v) =>
    v ? "YA" : "TIDAK"

f_bias_txt(int b) =>
    b > 0 ? "BULLISH" : b < 0 ? "BEARISH" : "NETRAL"

f_pip_size() =>
    string t = syminfo.ticker
    bool isJpy = str.contains(t, "JPY")
    bool isXau = str.contains(t, "XAU") or str.contains(t, "GOLD")
    bool isXag = str.contains(t, "XAG") or str.contains(t, "SILVER")
    isJpy ? 0.01 : isXau ? 0.10 : isXag ? 0.01 : 0.0001

f_to_pips(float priceDist) =>
    math.abs(priceDist) / f_pip_size()

f_bias(float c, float eFast, float eMid, float eTrend) =>
    c > eFast and eFast > eMid and eMid > eTrend ? 1 : c < eFast and eFast < eMid and eMid < eTrend ? -1 : c > eMid and eMid > eTrend ? 1 : c < eMid and eMid < eTrend ? -1 : 0

f_score_part(bool cond, int weight) =>
    cond ? weight : 0

f_line_update(line ln, float price, color col, style, int width, int barsForward) =>
    line out = ln
    if na(out)
        out := line.new(bar_index, price, bar_index + barsForward, price, xloc=xloc.bar_index, extend=extend.none, color=col, style=style, width=width)
    else
        line.set_xy1(out, bar_index, price)
        line.set_xy2(out, bar_index + barsForward, price)
        line.set_color(out, col)
        line.set_style(out, style)
        line.set_width(out, width)
    out

f_line_hide(line ln) =>
    if not na(ln)
        line.set_xy1(ln, bar_index, close)
        line.set_xy2(ln, bar_index, close)
        line.set_color(ln, color.new(color.gray, 100))
    ln

f_label_update(label lb, float price, string txt, color bg, color fg) =>
    label out = lb
    if na(out)
        out := label.new(bar_index, price, txt, xloc=xloc.bar_index, style=label.style_label_left, color=bg, textcolor=fg, size=size.small)
    else
        label.set_xy(out, bar_index, price)
        label.set_text(out, txt)
        label.set_color(out, bg)
        label.set_textcolor(out, fg)
    out

pivL = ta.pivotlow(low, pivotLeft, pivotRight)

if not na(pivH)
    lastSwingHigh := pivH
if not na(pivL)
    lastSwingLow := pivL

fallbackHigh = ta.highest(high, structureLen)[1]
fallbackLow  = ta.lowest(low, structureLen)[1]
refHigh = na(lastSwingHigh) ? fallbackHigh : lastSwingHigh
refLow  = na(lastSwingLow) ? fallbackLow : lastSwingLow

bosLong  = not na(refHigh) and close > refHigh and close[1] <= refHigh
bosShort = not na(refLow) and close < refLow and close[1] >= refLow
chochLong  = ltfBias <= 0 and bosLong
chochShort = ltfBias >= 0 and bosShort

prevLowest = ta.lowest(low[1], liqLookback)
prevHighest = ta.highest(high[1], liqLookback)
sweepLow = low < prevLowest and close > prevLowest
sweepHigh = high > prevHighest and close < prevHighest

eqHigh = math.abs(high - prevHighest) <= atr * equalTolATR
eqLow = math.abs(low - prevLowest) <= atr * equalTolATR

// SESSIONS

inAsia = not na(time(timeframe.period, asiaSession, sessionTz))

inNY = not na(time(timeframe.period, nySession, sessionTz))
inRollover = not na(time(timeframe.period, rolloverSess, sessionTz))
inOverlap = inLondon and inNY
sessionOK = not useSession or ((inAsia or inLondon or inNY) and not (avoidRollover and inRollover))
sessionName = inOverlap ? "LONDON+NY" : inLondon ? "LONDON" : inNY ? "NEW YORK" : inAsia ? "ASIA" : "OFF"
sessionMultiplier = inOverlap ? 1.15 : inLondon ? 1.08 : inNY ? 1.05 : inAsia ? 0.88 : 0.80

//==================================================
// FILTERS
//==================================================
trendOK = adx >= minADX and adx <= maxADX

atrTooHigh = timeframe.isintraday and atrPct > maxAtrPct
volOK = not useVolFilter or volRatio >= minVolRatio
canLong = tradeDirection != "SHORT ONLY"
canShort = tradeDirection != "LONG ONLY"

htfBlockLong = blockCounterHTF and htfMaxScore > 0 and htfShortScore == htfMaxScore
htfBlockShort = blockCounterHTF and htfMaxScore > 0 and htfLongScore == htfMaxScore

//==================================================
// SCORING ENGINE FOR FOREX
//==================================================
trendLong = close > emaFast and emaFast > emaMid and emaMid > emaTrend
trendShort = close < emaFast and emaFast < emaMid and emaMid < emaTrend
momentumLong = rsi > 52 and macdHist > 0 and macdLine > macdSig
momentumShort = rsi < 48 and macdHist < 0 and macdLine < macdSig
dmiLong = diPlus > diMinus and trendOK
dmiShort = diMinus > diPlus and trendOK
pullbackLong = low <= emaFast or low <= emaMid or close <= emaFast + atr * 0.35
pullbackShort = high >= emaFast or high >= emaMid or close >= emaFast - atr * 0.35
structureLong = bosLong or chochLong or sweepLow or close > refHigh
structureShort = bosShort or chochShort or sweepHigh or close < refLow
liquidityLong = sweepLow or eqLow
liquidityShort = sweepHigh or eqHigh

longScoreRaw = 0
longScoreRaw += f_score_part(trendLong, 22)
longScoreRaw += f_score_part(close > emaTrend, 8)
longScoreRaw += f_score_part(momentumLong, 16)
longScoreRaw += f_score_part(dmiLong, 14)
longScoreRaw += f_score_part(structureLong, 14)
longScoreRaw += f_score_part(pullbackLong or entryMode == "BREAKOUT", 8)
longScoreRaw += f_score_part(liquidityLong, 6)
longScoreRaw += htfMaxScore > 0 ? int(math.round(12.0 * htfLongScore / htfMaxScore)) : 6
longScoreRaw += f_score_part(sessionOK, 6)
longScoreRaw += f_score_part(volOK, 4)

shortScoreRaw = 0
shortScoreRaw += f_score_part(trendShort, 22)
shortScoreRaw += f_score_part(close < emaTrend, 8)
shortScoreRaw += f_score_part(momentumShort, 16)
shortScoreRaw += f_score_part(dmiShort, 14)
shortScoreRaw += f_score_part(structureShort, 14)
shortScoreRaw += f_score_part(pullbackShort or entryMode == "BREAKOUT", 8)
shortScoreRaw += f_score_part(liquidityShort, 6)
shortScoreRaw += htfMaxScore > 0 ? int(math.round(12.0 * htfShortScore / htfMaxScore)) : 6
shortScoreRaw += f_score_part(sessionOK, 6)
shortScoreRaw += f_score_part(volOK, 4)

penalty = 0
penalty += chop ? 12 : 0
penalty += atrTooHigh ? 10 : 0
penalty += inRollover and avoidRollover ? 15 : 0
penalty += adx > maxADX ? 8 : 0

longScore = int(f_clamp(longScoreRaw - penalty - (htfBlockLong ? 18 : 0), 0, 100))
shortScore = int(f_clamp(shortScoreRaw - penalty - (htfBlockShort ? 18 : 0), 0, 100))

longValid = canLong and sessionOK and volOK and not atrTooHigh and not htfBlockLong and longScore >= minScore and longScore >= shortScore + minScoreGap
shortValid = canShort and sessionOK and volOK and not atrTooHigh and not htfBlockShort and shortScore >= minScore and shortScore >= longScore + minScoreGap

int signalDir = longValid ? 1 : shortValid ? -1 : 0
int dominantDir = longScore > shortScore ? 1 : shortScore > longScore ? -1 : 0
activeScore = signalDir == 1 ? longScore : signalDir == -1 ? shortScore : math.max(longScore, shortScore)

//==================================================
// ENTRY, SL, TP ENGINE
//==================================================
spreadBuffer = manualSpreadPips * f_pip_size()

longRawEntry = entryMode == "MARKET" ? close : entryMode == "BREAKOUT" ? math.max(close, refHigh + atr * 0.03) : close - atr * entryATR
shortRawEntry = entryMode == "MARKET" ? close : entryMode == "BREAKOUT" ? math.min(close, refLow - atr * 0.03) : close + atr * entryATR

longEntry = math.min(close, longRawEntry)
shortEntry = math.max(close, shortRawEntry)
longZoneLo = longEntry - atr * entryZoneATR
longZoneHi = longEntry + atr * entryZoneATR
shortZoneLo = shortEntry - atr * entryZoneATR
shortZoneHi = shortEntry + atr * entryZoneATR

longSLStruct = math.min(refLow, ta.lowest(low, structureLen)) - atr * slStructATR - spreadBuffer
shortSLStruct = math.max(refHigh, ta.highest(high, structureLen)) + atr * slStructATR + spreadBuffer
longSLAtr = longEntry - atr * slATR - spreadBuffer
shortSLAtr = shortEntry + atr * slATR + spreadBuffer
longSL = math.min(longSLAtr, longSLStruct)
shortSL = math.max(shortSLAtr, shortSLStruct)

longRisk = math.max(longEntry - longSL, atr * 0.35)
shortRisk = math.max(shortSL - shortEntry, atr * 0.35)

longTP1 = longEntry + longRisk * rrTP1 * sessionMultiplier
longTP2 = longEntry + longRisk * rrTP2 * sessionMultiplier
longTP3 = longEntry + longRisk * rrTP3 * sessionMultiplier
shortTP1 = shortEntry - shortRisk * rrTP1 * sessionMultiplier
shortTP2 = shortEntry - shortRisk * rrTP2 * sessionMultiplier
shortTP3 = shortEntry - shortRisk * rrTP3 * sessionMultiplier

entry = signalDir == 1 ? longEntry : signalDir == -1 ? shortEntry : dominantDir == 1 ? longEntry : shortEntry
zoneLo = signalDir == 1 ? longZoneLo : signalDir == -1 ? shortZoneLo : dominantDir == 1 ? longZoneLo : shortZoneLo
zoneHi = signalDir == 1 ? longZoneHi : signalDir == -1 ? shortZoneHi : dominantDir == 1 ? longZoneHi : shortZoneHi
sl = signalDir == 1 ? longSL : signalDir == -1 ? shortSL : dominantDir == 1 ? longSL : shortSL
tp1 = signalDir == 1 ? longTP1 : signalDir == -1 ? shortTP1 : dominantDir == 1 ? longTP1 : shortTP1
tp2 = signalDir == 1 ? longTP2 : signalDir == -1 ? shortTP2 : dominantDir == 1 ? longTP2 : shortTP2
tp3 = signalDir == 1 ? longTP3 : signalDir == -1 ? shortTP3 : dominantDir == 1 ? longTP3 : shortTP3
riskNow = signalDir == 1 ? longRisk : signalDir == -1 ? shortRisk : dominantDir == 1 ? longRisk : shortRisk

rrLive = signalDir == 1 ? f_safe_div(tp1 - entry, entry - sl, 0.0) : signalDir == -1 ? f_safe_div(entry - tp1, sl - entry, 0.0) : na
prediction = signalDir == 1 ? "BUY / NAIK" : signalDir == -1 ? "SELL / TURUN" : dominantDir == 1 ? "WAIT BUY" : dominantDir == -1 ? "WAIT SELL" : "WAIT"
status = signalDir != 0 ? "VALID" : chop ? "CHOP / TUNGGU" : atrTooHigh ? "ATR TERLALU TINGGI" : not sessionOK ? "DI LUAR SESI" : "MENUNGGU KONFIRMASI"
