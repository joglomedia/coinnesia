//==================================================
// FUNCTIONS
//==================================================
f_clamp(x, lo, hi) => math.max(lo, math.min(x, hi))
f_fmt(x) => str.tostring(x, format.mintick)
f_fmt_safe(x) => na(x) ? "-" : str.tostring(x, format.mintick)
f_bias(c, e20, e50) => c > e20 and e20 > e50 ? "BELI" : c < e20 and e20 < e50 ? "JUAL" : "SIDEWAYS"
f_bias_score(b) => b == "BELI" ? 1 : b == "JUAL" ? -1 : 0
f_bias_num(c, e20, e50) => c > e20 and e20 > e50 ? 1 : c < e20 and e20 < e50 ? -1 : 0
f_reach_long(rawLevel, price, atrVal, minATR, maxATR) =>
    dist = atrVal > 0.0 ? (price - rawLevel) / atrVal : minATR
    price - atrVal * f_clamp(dist, minATR, maxATR)
f_reach_short(rawLevel, price, atrVal, minATR, maxATR) =>
    dist = atrVal > 0.0 ? (rawLevel - price) / atrVal : minATR
    price + atrVal * f_clamp(dist, minATR, maxATR)
f_cap_long(level, base, atrVal, maxATR) => math.min(level, base + atrVal * maxATR)
f_cap_short(level, base, atrVal, maxATR) => math.max(level, base - atrVal * maxATR)
f_session_extra(sess) => sess == "ASIA" ? slAsiaATR : sess == "EROPA" ? slEuropeATR : slUSAATR
f_session_from_minutes(tMin) => tMin >= 420 and tMin < 900 ? "ASIA" : tMin >= 900 and tMin < 1230 ? "EROPA" : "USA"
f_session_from_ts(ts) =>
    int _h = hour(ts, WIB)
    int _m = minute(ts, WIB)
    f_session_from_minutes(_h * 60 + _m)
f_session_time_range(sess) => sess == "ASIA" ? "07:00-14:59 WIB" : sess == "EROPA" ? "15:00-20:29 WIB" : "20:30-02:59 WIB"
f_long_sl(baseStop, deepAdd, swingLow, vwapRef, emaRef, atrVal, sess, trapNow, coolNow, volShockNow, lowerWick) =>
    sessPad = atrVal * f_session_extra(sess)
    trapPad = trapNow or coolNow ? atrVal * slTrapExtraATR : 0.0
    wickPad = lowerWick >= atrVal * wickATRTrap ? atrVal * slWickExtraATR : 0.0
    volPad  = volShockNow ? atrVal * slVolExtraATR : 0.0
    pad = sessPad + trapPad + wickPad + volPad
    structStop = math.min(swingLow - pad, math.min(vwapRef, emaRef) - pad * 0.50)
    deepStop = deepAdd - atrVal * (minSLDistanceATR + f_session_extra(sess))
    math.min(baseStop, math.min(structStop, deepStop))
f_short_sl(baseStop, deepAdd, swingHigh, vwapRef, emaRef, atrVal, sess, trapNow, coolNow, volShockNow, upperWick) =>
    sessPad = atrVal * f_session_extra(sess)
    trapPad = trapNow or coolNow ? atrVal * slTrapExtraATR : 0.0
    wickPad = upperWick >= atrVal * wickATRTrap ? atrVal * slWickExtraATR : 0.0
    volPad  = volShockNow ? atrVal * slVolExtraATR : 0.0
    pad = sessPad + trapPad + wickPad + volPad
    structStop = math.max(swingHigh + pad, math.max(vwapRef, emaRef) + pad * 0.50)
    deepStop = deepAdd + atrVal * (minSLDistanceATR + f_session_extra(sess))
    math.max(baseStop, math.max(structStop, deepStop))
f_long_tp_fix(tpRaw, base, prev, atrVal) => math.max(tpRaw, math.max(base + atrVal * tpStepMinATR, prev + atrVal * tpStepMinATR))

f_long_tp_flow(tpRaw, base, prev, atrVal, maxDistATR, minStepATR) => math.max(math.min(tpRaw, base + atrVal * maxDistATR), math.max(base + atrVal * minStepATR, prev + atrVal * minStepATR))
f_short_tp_flow(tpRaw, base, prev, atrVal, maxDistATR, minStepATR) => math.min(math.max(tpRaw, base - atrVal * maxDistATR), math.min(base - atrVal * minStepATR, prev - atrVal * minStepATR))
f_bool_text(x) => x ? "OK" : "NO"

f_2(n) =>

f_valid_ts(ts) =>
    not na(ts) and ts > 946684800000

f_ts_wib(ts) =>
    f_valid_ts(ts) ? str.tostring(dayofmonth(ts, WIB)) + " " + f_mon(month(ts, WIB)) + " " + f_2(hour(ts, WIB)) + ":" + f_2(minute(ts, WIB)) : "ETA INVALID"

f_tf_millis() =>
    f_safe_tf_millis()

// ETA must remain visible even when trade is not executable.
// All timestamps are forced into the future; no panel time may point to a passed period.
f_future_anchor(tfMs) =>
    int anchor = f_valid_ts(timenow) ? timenow : time
    if f_valid_ts(time) and tfMs > 0
        int elapsed = anchor <= time ? 0 : int(math.ceil((anchor - time) / tfMs))
        int safeElapsed = math.max(elapsed, 0)
        int nextFromBar = time + safeElapsed * tfMs
        nextFromBar <= anchor ? nextFromBar + tfMs : nextFromBar
    else
        anchor + math.max(tfMs, 3600000)

f_eta_txt(level, origin, enabled) =>
    float movePerBarRaw = ta.sma(math.abs(close - close[1]), 14)
    float movePerBar = math.max(nz(movePerBarRaw, atr), syminfo.mintick)
    int tfMs = f_tf_millis()
    bool validInput = not na(level) and not na(origin) and not na(movePerBar) and movePerBar > 0 and tfMs > 0 and f_valid_ts(time)
    if not validInput
        "ETA INVALID"
    else
        float barsRaw = math.abs(level - origin) / movePerBar
        barsRaw := f_clamp(barsRaw, 1.0, 500.0)
        int barsLo = math.max(1, int(math.floor(barsRaw * 0.85)))
        int barsHi = math.max(barsLo, int(math.ceil(barsRaw * 1.20)))
        int baseTs = f_future_anchor(tfMs)
        int t1 = baseTs + int(math.max(barsLo - 1, 0) * tfMs)
        int t2 = baseTs + int(math.max(barsHi - 1, 0) * tfMs)
        if f_valid_ts(t1) and f_valid_ts(timenow) and t1 <= timenow
            t1 := t1 + tfMs
            t2 := math.max(t2 + tfMs, t1)
        f_ts_wib(t1) + " - " + f_ts_wib(t2)
